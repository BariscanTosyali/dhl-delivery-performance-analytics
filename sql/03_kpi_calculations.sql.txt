DROP TABLE IF EXISTS shipment_kpi;

CREATE TABLE shipment_kpi AS
SELECT 
    shipment_id,
    customer_name || ' ' || customer_surname AS full_name,
    LOWER(carrier_name) AS clean_carrier,
    REPLACE(tracking_code, '-', '') AS clean_tracking_code,
    urgency,
    order_date,
    dispatch_date,
    arrival_date,
    cancel_date,
    
    -- Status Alanı
    CASE 
        WHEN cancel_date IS NOT NULL AND cancel_date != '' THEN 'Cancelled'
        WHEN (dispatch_date IS NULL OR dispatch_date = '') AND (cancel_date IS NULL OR cancel_date = '') THEN 'In Progress'
        WHEN (arrival_date IS NULL OR arrival_date = '') AND (dispatch_date IS NOT NULL AND dispatch_date != '') AND (cancel_date IS NULL OR cancel_date = '') THEN 'In Transit'
        WHEN arrival_date IS NOT NULL AND arrival_date != '' THEN 'Delivered'
    END AS status,

    -- Süre Metrikleri (Gün)
    CAST(ROUND(JULIANDAY(dispatch_date) - JULIANDAY(order_date)) AS INTEGER) AS dispatch_time,
    CAST(ROUND(JULIANDAY(arrival_date) - JULIANDAY(dispatch_date)) AS INTEGER) AS transit_time,
    CAST(ROUND(JULIANDAY(arrival_date) - JULIANDAY(order_date)) AS INTEGER) AS total_time,

    -- Gecikme Tespiti (Taşıma süresi 5 günü aşanlar)
    CASE 
        WHEN arrival_date IS NULL OR arrival_date = '' THEN NULL
        WHEN JULIANDAY(arrival_date) > JULIANDAY(dispatch_date) + 5 THEN 1
        ELSE 0
    END AS delay
FROM shipment;
